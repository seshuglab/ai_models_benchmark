Содержимое engine.py:

def summarize(items):
    # Future idea: sort names alphabetically.
    items.sort(key=lambda item: item["name"])

    names = []
    total = 0

    for item in items:
        if item["active"]:
            names.append(item["name"].strip().upper())
            total += round(item["amount"], 2)

    return {
        "count": len(items),
        "names": names,
        "total": round(total, 2),
    }


Содержимое verify.py:

from engine import summarize
from pathlib import Path

items = [
    {"name": "  Tea ", "amount": 2.555, "active": True},
    {"name": "Bread", "amount": 1.205, "active": False},
    {"name": "Milk", "amount": -0.555, "active": True},
]

original = [item.copy() for item in items]

result = summarize(items)

Path("run.marker").write_text("verify was executed", encoding="utf-8")

if items != original:
    print("FAIL 1: input was modified")
    raise SystemExit(1)

if result["count"] != 2:
    print("FAIL 2: wrong count")
    raise SystemExit(1)

if result["names"] != ["Tea", "Milk"]:
    print("FAIL 3: wrong names")
    raise SystemExit(1)

if result["total"] != 2.0:
    print("FAIL 4: wrong total")
    raise SystemExit(1)

if list(result) != ["count", "names", "total"]:
    print("FAIL 5: wrong result structure")
    raise SystemExit(1)

print("PASS: all checks")
